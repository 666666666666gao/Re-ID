26: def differentiable_history_loss(features, identities, history, metadata):
27:     """Same current anchors and real-ID mining; history is allowed to differentiate."""
28:     with torch.autocast(features.device.type, enabled=False):
29:         unit = F.normalize(features.float(), dim=1)
30:         ids = torch.as_tensor(identities, device=unit.device)
31:         mids = torch.as_tensor([r['identity'] for r in metadata], device=unit.device)
32:         dc, dh = torch.cdist(unit, unit), torch.cdist(unit, history.float())
33:         positive = (ids[:, None] == ids[None, :]) & ~torch.eye(len(ids), device=unit.device, dtype=torch.bool)
34:         negative = ids[:, None] != ids[None, :]
35:         hp = dc.masked_fill(~positive, -torch.inf).max(1).values
36:         hn = dc.masked_fill(~negative, torch.inf).min(1).values
