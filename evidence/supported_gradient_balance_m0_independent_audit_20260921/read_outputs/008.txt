223: def scene_scores(distances, query_ids, gallery_ids, query_scenes, gallery_scenes):
224:     import numpy as np
225: 
226:     ap, first = [], []
227:     for index, order in enumerate(np.argsort(distances, axis=1)):
228:         keep = ~((gallery_ids[order] == query_ids[index]) &
229:                  (gallery_scenes[order] == query_scenes[index]))
230:         matches = gallery_ids[order][keep] == query_ids[index]
231:         positions = np.flatnonzero(matches)
232:         assert positions.size
233:         ap.append(float(np.mean(np.cumsum(matches)[positions] / (positions + 1))))
234:         first.append(int(positions[0] + 1))
235:     metrics = {"mAP": float(np.mean(ap) * 100)}
236:     metrics.update({f"Rank-{rank}": float(np.mean(np.asarray(first) <= rank) * 100)
237:                     for rank in (1, 5, 10)})
238:     return {"metrics": metrics, "average_precision": ap, "first_match_rank": first}
