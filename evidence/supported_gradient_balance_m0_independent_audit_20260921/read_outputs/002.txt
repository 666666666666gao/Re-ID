11: def cross_scene_ap_from_distances(current, historical, identities,
12:                                   history_identities, scenes, history_scenes):
13:     """Return eligible-anchor mean loss, per-anchor AP, and positive counts.
14: 
15: AP=0 is a storage placeholder for ineligible anchors; it is excluded from
16: the loss mean. Whole batches without eligible anchors actually occur in the
17: registered source sequence and return a graph-connected exact zero.
18: """
19:     batch = current.shape[0]
20:     assert current.shape == (batch, batch) and historical.shape[0] == batch
21:     assert current.dtype in (torch.float32, torch.float64)
22:     assert historical.dtype == current.dtype and historical.device == current.device
23:     ids = torch.as_tensor(identities, device=current.device)
24:     mids = torch.as_tensor(history_identities, device=current.device, dtype=ids.dtype)
25:     env = torch.as_tensor(scenes, device=current.device)
26:     menv = torch.as_tensor(history_scenes, device=current.device, dtype=env.dtype)
27:     assert len(ids) == len(env) == batch
28:     assert len(mids) == len(menv) == historical.shape[1]
29:     with torch.autocast(current.device.type, enabled=False):
30:         distance = torch.cat((current, historical), dim=1)
31:         score = 1 - distance.square() / 2
32:         same_id = ids[:, None] == torch.cat((ids, mids))[None, :]
33:         same_scene = env[:, None] == torch.cat((env, menv))[None, :]
34:         positive = same_id & ~same_scene
35:         valid = ~(same_id & same_scene)
36:         counts = positive.sum(1)
37:         eligible = counts > 0
38:         assert (~same_id).any(1).all()
39:         positions = torch.arange(distance.shape[1], device=current.device)
40:         aps = []
41:         for i in range(batch):
42:             pos = positions[positive[i]]
43:             if pos.numel() == 0:
44:                 aps.append(score[i].sum() * 0)
45:                 continue
46:             comparisons = torch.sigmoid((score[i][None, :] - score[i, pos][:, None]) / TAU)
47:             comparison_valid = valid[i][None, :] & (positions[None, :] != pos[:, None])
48:             comparisons = comparisons.masked_fill(~comparison_valid, 0)
49:             positive_rank = 1 + (comparisons * positive[i][None, :]).sum(1)
50:             all_rank = 1 + comparisons.sum(1)
51:             aps.append((positive_rank / all_rank).mean())

53:         loss = 1 - per_anchor[eligible].mean() if bool(eligible.any()) else distance.sum() * 0
