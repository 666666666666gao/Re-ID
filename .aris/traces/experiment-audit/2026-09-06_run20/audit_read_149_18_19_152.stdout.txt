19: def sha256(path):
20:     digest = hashlib.sha256()
21:     with Path(path).open("rb") as handle:
22:         for chunk in iter(lambda: handle.read(1024 * 1024), b""):
23:             digest.update(chunk)
24:     return digest.hexdigest()

141: def train_source(model, loader, cfg, record_indices, records, *, preflight, step_log):
142:     import numpy as np
143:     import torch
144:     from layers.make_loss import make_loss
145:     from solver.make_optimizer import make_optimizer
146:     from solver.scheduler_factory import create_scheduler
147:     from tools.build_v12_complete_path_oof_targets import _signal_training_loss
148:     from tools.run_signal_preserving_v5 import _module_state_sha256
149: 
150:     loss_fn, center = make_loss(cfg, num_classes=model.num_classes)
151:     optimizer, _ = make_optimizer(cfg, model, center)
152:     scheduler = create_scheduler(cfg, optimizer)
153:     scaler = torch.amp.GradScaler("cuda", init_scale=256.0)
154:     names = {id(p): name for name, p in model.named_parameters()}
155:     optimizer_groups = [{"name": names[id(group["params"][0])], "lr": group["lr"],
156:                          "initial_lr": group["initial_lr"],
157:                          "weight_decay": group["weight_decay"]}
158:                         for group in optimizer.param_groups]
159:     name_to_index = {Path(row[0]).name: index
160:                      for row, index in zip(records, record_indices, strict=True)}
161:     assert len(name_to_index) == len(records)
162:     initial_state = _module_state_sha256(model)
163:     selector_initial_state = _module_state_sha256(model.SIM.token_selection)
164:     trainable_names = {name for name, p in model.named_parameters() if p.requires_grad}
165:     gradient_names = set()
166:     history, steps = [], []
167:     for epoch in range(1, (1 if preflight else 30) + 1):
168:         started = time.perf_counter()
169:         scheduler.step(epoch)
170:         model.train()
171:         epoch_losses = []
172:         for images, labels, cameras, scenes, paths in loader:
173:             assert labels.numel() == 64
174:             assert all(tuple(value.shape) == (64, 3, 128, 256) for value in images.values())
175:             assert sorted(torch.unique(labels, return_counts=True)[1].tolist()) == [8] * 8
176:             optimizer.zero_grad(set_to_none=True)
177:             images = {key: value.cuda(non_blocking=True) for key, value in images.items()}
178:             labels, cameras, scenes = (x.cuda(non_blocking=True) for x in (labels, cameras, scenes))
179:             components = []
180: 
181:             def recorded_loss(**kwargs):
182:                 component = loss_fn(**kwargs)
183:                 components.append(float(component.detach()))
184:                 return component
185: 
186:             with torch.autocast("cuda", dtype=torch.float16):
187:                 output = model(images, label=labels, cam_label=cameras, view_label=scenes,
188:                                training=True, sge=cfg.MODEL.stageName)
189:                 assert output[0] == 3 and len(output) == 11
190:                 loss = _signal_training_loss(
191:                     output, loss_fn=recorded_loss, labels=labels, cameras=cameras,
192:                     stage=cfg.MODEL.stageName, gram_weight=cfg.MODEL.Gram_Loss_weight,
193:                     patch_weight=cfg.MODEL.PAT_Loss_weight)
194:             assert torch.isfinite(loss).item()
195:             scale_before = scaler.get_scale()
196:             scaler.scale(loss).backward()
197:             scaler.unscale_(optimizer)
198:             for name, parameter in model.named_parameters():
199:                 if parameter.grad is not None:
200:                     gradient_names.add(name)
201:             scaler.step(optimizer)
202:             scaler.update()
203:             row = {"step": len(steps) + 1, "epoch": epoch, "loss": float(loss.detach()),
204:                    "id_triplet_head_losses": components, "gram_loss": float(output[-2].detach()),
205:                    "patch_loss": float(output[-1].detach()),
206:                    "sampled_record_indices": [name_to_index[p] for p in paths],
207:                    "amp_scale_before": scale_before, "amp_scale_after": scaler.get_scale(),
208:                    "optimizer_update_applied": scaler.get_scale() >= scale_before}
209:             with Path(step_log).open("a", encoding="utf-8") as handle:
210:                 handle.write(json.dumps(row, allow_nan=False) + "\n")
211:             assert row["optimizer_update_applied"], "AMP overflow; fixed run stops"
212:             if preflight:
213:                 for name, parameter in model.named_parameters():
214:                     if parameter.grad is not None:
215:                         assert torch.isfinite(parameter.grad).all().item(), name
216:             steps.append(row)
217:             epoch_losses.append(row["loss"])
218:         history.append({"epoch": epoch, "optimizer_steps": len(epoch_losses),
219:                         "mean_loss": float(np.mean(epoch_losses)),
220:                         "learning_rates": sorted({group["lr"] for group in optimizer.param_groups}),
221:                         "elapsed_seconds": time.perf_counter() - started})
222:         print(json.dumps({"event": "signal_source_epoch", **history[-1]}), flush=True)
223:     selector_final_state = _module_state_sha256(model.SIM.token_selection)
224:     assert selector_final_state == selector_initial_state
225:     return {"epochs": len(history), "optimizer_steps": len(steps), "overflow_events": 0,
226:             "frozen_token_selection_initial_sha256": selector_initial_state,
227:             "frozen_token_selection_final_sha256": selector_final_state,
228:             "frozen_token_selection_parameters": sum(p.numel() for p in model.SIM.token_selection.parameters()),
229:             "initial_state_sha256": initial_state, "final_state_sha256": _module_state_sha256(model),
230:             "total_parameters": sum(p.numel() for p in model.parameters()),
231:             "trainable_parameters": sum(p.numel() for p in model.parameters() if p.requires_grad),
232:             "trainable_tensors": len(trainable_names), "gradient_tensors": len(gradient_names),
233:             "trainable_without_gradient": sorted(trainable_names - gradient_names),
234:             "optimizer_groups": optimizer_groups, "history": history, "steps": steps}

253: def camera_scores(distances, query_ids, gallery_ids, query_cameras, gallery_cameras):
254:     import numpy as np
255: 
256:     ap, first = [], []
257:     for index, order in enumerate(np.argsort(distances, axis=1)):
258:         keep = ~((gallery_ids[order] == query_ids[index]) &
259:                  (gallery_cameras[order] == query_cameras[index]))
260:         matches = gallery_ids[order][keep] == query_ids[index]
261:         positions = np.flatnonzero(matches)
262:         assert positions.size
263:         ap.append(float(np.mean(np.cumsum(matches)[positions] / (positions + 1))))
264:         first.append(int(positions[0] + 1))
265:     metrics = {"mAP": float(np.mean(ap) * 100)}
266:     metrics.update({f"Rank-{rank}": float(np.mean(np.asarray(first) <= rank) * 100)
267:                     for rank in (1, 5, 10)})
268:     return {"metrics": metrics, "average_precision": ap, "first_match_rank": first}

271: def evaluate_gallery(features, protocol, fold, directory):
272:     import numpy as np
273:     import torch
274:     from utils.metrics import eval_func
275: 
276:     rows = [protocol["records"][i] for i in fold["gallery_record_indices"]]
277:     positions = [row["gallery_position"] for row in fold["query_rows"]]
278:     for query, position in zip(fold["query_rows"], positions, strict=True):
279:         assert rows[position]["index"] == query["record_index"]
280:     ids = np.asarray([r["identity"] for r in rows])
281:     cameras = np.asarray([r["camera"] for r in rows])
282:     normalized = torch.nn.functional.normalize(features.float(), dim=1)
283:     qf, gf = normalized[positions], normalized
284:     distances = qf.square().sum(1, keepdim=True) + gf.square().sum(1)[None]
285:     distances.addmm_(qf, gf.T, beta=1, alpha=-2)
286:     scores = camera_scores(distances.numpy(), ids[positions], ids, cameras[positions], cameras)
287:     cmc, mean_ap = eval_func(distances.numpy(), ids[positions], ids, cameras[positions], cameras)
288:     differences = {"mAP": abs(scores["metrics"]["mAP"] - float(mean_ap * 100))}
289:     differences.update({f"Rank-{rank}": abs(scores["metrics"][f"Rank-{rank}"] -
290:                                            float(cmc[rank - 1] * 100)) for rank in (1, 5, 10)})
291:     assert differences["mAP"] < 1e-10 and max(differences.values()) < 1e-5
292:     torch.save({"features": features, "distances": distances,
293:                 "gallery_record_indices": fold["gallery_record_indices"],
294:                 "query_gallery_positions": positions}, directory / "retrieval_arrays.pt")
295:     ranking_path = directory / "full_rankings.json.gz"
296:     with gzip.open(ranking_path, "wt", encoding="utf-8") as handle:
297:         json.dump({"schema": "rgbnt100-complete-gallery-order-v1",
298:                    "gallery_record_indices": fold["gallery_record_indices"],
299:                    "query_gallery_positions": positions,
300:                    "ordered_gallery_positions": np.argsort(distances.numpy(), axis=1).tolist()},
301:                   handle, separators=(",", ":"))
302:     return {**scores, "upstream_metric_difference_pp": differences,
303:             "full_rankings_sha256": sha256(ranking_path),
304:             "gallery_manifest": rows, "query_rows": fold["query_rows"],
305:             "feature_width": 3072, "retrieval_arrays_sha256": sha256(directory / "retrieval_arrays.pt")}
