14: def volume_computation3(language, video, audio):
15: 
16:     """
17:     Computes the volume for each pair of samples between language (shape [batch_size1, feature_dim])
18:     and video, audio, subtitles (shape [batch_size2, feature_dim]) using the determinant of a 4x4
19:     Gram matrix.
20:     
21:     Parameters:
22:     - language (torch.Tensor): Tensor of shape (batch_size1, feature_dim) representing language features.
23:     - video (torch.Tensor): Tensor of shape (batch_size2, feature_dim) representing video features.
24:     - audio (torch.Tensor): Tensor of shape (batch_size2, feature_dim) representing audio features.
25:     - subtitles (torch.Tensor): Tensor of shape (batch_size2, feature_dim) representing subtitle features.
26:     
27:     Returns:
28:     - torch.Tensor: Tensor of shape (batch_size1, batch_size2) representing the volume for each pair.
29:     """
30: 
31:     batch_size1 = language.shape[0]  # For language
32:     batch_size2 = video.shape[0]     # For video, audio, subtitles
33: 
34:     # Compute pairwise dot products for language with itself (shape: [batch_size1, 1])
35:     ll = torch.einsum('bi,bi->b', language, language).unsqueeze(1).expand(-1, batch_size2)
36: 
37:     # Compute pairwise dot products for language with video, audio, and subtitles (shape: [batch_size1, batch_size2])
38:     lv = language@video.T  
39:     la = language@audio.T
40: 
41:     # Compute pairwise dot products for video, audio, and subtitles with themselves and with each other
42:     vv = torch.einsum('bi,bi->b', video, video).unsqueeze(0).expand(batch_size1, -1)
43:     va = torch.einsum('bi,bi->b', video, audio).unsqueeze(0).expand(batch_size1, -1)
44:     aa = torch.einsum('bi,bi->b', audio, audio).unsqueeze(0).expand(batch_size1, -1)
45:     
46: 
47: 
48:     # Stack the results to form the Gram matrix for each pair (shape: [batch_size1, batch_size2, 3, 3])
49:     G = torch.stack([
50:         torch.stack([ll, lv, la], dim=-1),  # First row of the Gram matrix
51:         torch.stack([lv, vv, va], dim=-1),  # Second row of the Gram matrix
52:         torch.stack([la, va, aa], dim=-1)  # Third row of the Gram matrix
53:     ], dim=-2)
54:     # G.shape [b1,b2,3,3]
55: 
56:     # Compute the determinant for each Gram matrix (shape: [batch_size1, batch_size2])
57:     gram_det = torch.det(G.float())
58: 
59:     # Compute the square root of the absolute value of the determinants
60:     res = torch.sqrt(torch.abs(gram_det))
61:     # print(res.shape)
62:     return res
